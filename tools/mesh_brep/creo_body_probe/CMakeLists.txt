cmake_minimum_required(VERSION 3.25)
project(creo_body_probe LANGUAGES CXX)
if(NOT CREO_ROOT)
    message(FATAL_ERROR "Pass -DCREO_ROOT=<Creo installation>")
endif()
set(TK "${CREO_ROOT}/Common Files/protoolkit")
add_executable(creo_body_probe ../creo_body_probe.cpp)
target_compile_features(creo_body_probe PRIVATE cxx_std_17)
target_include_directories(creo_body_probe PRIVATE "${TK}/includes")
target_link_libraries(creo_body_probe PRIVATE "${TK}/x86e_win64/obj/protkmd_NU.lib"
    "${TK}/x86e_win64/obj/ucore.lib" "${TK}/x86e_win64/obj/udata.lib" ws2_32 mpr netapi32)
if(MSVC)
    set_property(TARGET creo_body_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
    target_compile_definitions(creo_body_probe PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
